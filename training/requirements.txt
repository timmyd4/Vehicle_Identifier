# Python dependencies for the training half of car-identifier.
# (The app is Java/Spring Boot and has its own dependencies in app/pom.xml.)
#
# Install PyTorch FIRST, from PyTorch's own package index, so you get a build
# that matches your hardware. Plain `pip install torch` on Windows gives a
# CPU-only build that cannot use an NVIDIA GPU.
#
#   NVIDIA GPU (CUDA 13.0 build, needs a recent NVIDIA driver):
#     pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cu130
#
#   CPU only (no NVIDIA GPU, or just running the app):
#     pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cpu
#
# Then install everything else (from the project root):
#     pip install -r training/requirements.txt
#
# Versions are pinned (==) so the project installs the same way on every machine.

# --- Deep learning ---
torch==2.14.1             # tensors, GPU support, training loop building blocks
torchvision==0.29.1       # image transforms (resize, crop, augmentation)
timm==1.0.30              # library of pretrained image models (ConvNeXt, EfficientNet, ...)

# --- Data handling ---
pandas==3.1.0             # tables: reading/writing the train/val/test CSV files
scikit-learn==1.9.1       # stratified train/val/test split, confusion matrix
pillow==13.0.0            # opening and decoding image files
pyyaml==6.0.3             # reading the config.yaml settings file

# --- Progress and plots ---
tqdm==4.70.1              # progress bars for long loops
matplotlib==3.11.2        # charts (class distribution, training curves)
